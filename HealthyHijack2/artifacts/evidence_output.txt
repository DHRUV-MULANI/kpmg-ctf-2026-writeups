# /etc/crontab: system-wide crontab
# Unlike any other crontab you don't have to run the `crontab'
# command to install the new version when you edit this file

# that none of the other crontabs do.

PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# .---------------- minute (0 - 59)

# *  *  *  *  * user-name command to be executed

### [4] entrypoint.sh (flag creation + cron)

#!/bin/sh

# Generate defaults separately
DEFAULT_USER_FLAG="KPMG_CTF{$(head -c 16 /dev/urandom | xxd -p)}"
DEFAULT_ROOT_FLAG="KPMG_CTF{$(head -c 16 /dev/urandom | xxd -p)}"

# Use provided env vars or defaults
USER_FLAG="${USER_FLAG:-$DEFAULT_USER_FLAG}"
ROOT_FLAG="${ROOT_FLAG:-$DEFAULT_ROOT_FLAG}"

# Write user flag
mkdir -p /home/appuser

echo "$USER_FLAG" > /home/appuser/user.txt

chown appuser:appuser /home/appuser/user.txt
chmod 644 /home/appuser/user.txt

# Write root flag
mkdir -p /root

echo "$ROOT_FLAG" > /root/root.txt
chmod 600 /root/root.txt

# Cleanup env vars
unset USER_FLAG
unset ROOT_FLAG

# Cleanup SUID binaries planted by players (every 1 minute)
echo '* * * * * root find /tmp -perm /4000 -delete' >> /etc/crontab


# Start cron
service cron start

# Run app
exec su appuser -c "python /app/app.py"
