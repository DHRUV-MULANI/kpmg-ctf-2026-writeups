$ cat /usr/local/bin/acme-fs-guard.sh 2>&1
#!/bin/bash
# Continuously reconciles /home/maya.chen back to its baseline (captured
# once at setup time), deleting anything a player session creates shortly
# after it appears. Files only -- never touches processes -- so it cannot
# interrupt a live shell/SSH/python session. Runs as its own supervisord
# program, independent of any player connection.
set -u

HOME_DIR=/home/maya.chen
BASELINE=/var/lib/acme/.maya-home-baseline.list
INTERVAL="${CLEANUP_INTERVAL_SECONDS:-2}"

while true; do
    if [ -d "$HOME_DIR" ] && [ -f "$BASELINE" ]; then
        while IFS= read -r path; do
            rel="${path#$HOME_DIR/}"
            if [ "$path" = "$rel" ]; then
                continue  # malformed entry, skip rather than risk misinterpreting it
            fi
            if grep -Fxq "$rel" "$BASELINE"; then
                continue  # part of the original challenge install
            fi
            if [ -L "$path" ]; then
                rm -f "$path" 2>/dev/null && echo "$(date -Iseconds) removed symlink: $rel"
            elif [ -d "$path" ]; then
                # only removes if now empty; a non-empty new directory is
                # caught on a later pass once its contents are gone (find
                # -depth below already lists children before parents, so
                # this is normally already empty by the time we reach it)
                rmdir "$path" 2>/dev/null && echo "$(date -Iseconds) removed dir: $rel"
            else
                rm -f "$path" 2>/dev/null && echo "$(date -Iseconds) removed file: $rel"
            fi
        done < <(find "$HOME_DIR" -mindepth 1 -depth 2>/dev/null)
    fi
    sleep "$INTERVAL"
done
